	for(set @i, @x1; @i <= @x2; set @i, @i +1){
		for(set @j, @y1; @j <= @y2; set @j, @j+1){
			del @mob;
			del @type;
			set @mob, get_mobxy(@i, @j);
			if(!@mob){goto go;}
			set @type, istype(@mob);
			if(@mob && @type == 1){
				damaged @mob, @damage;
    				effect @mob, 0, 72, 75;
			}else{
			go:
				if(get_map_pk() && istype(get_char_serial(@myid,@i,@j)) == 3){
					set @mob, get_char_serial(@myid,@i,@j);
					if(@myid != @mob){
						char_damaged2 @mob, @damage,@damage;
						effect @mob, 0, 72, 75;
					}
				}
			}
		}
	}
}
