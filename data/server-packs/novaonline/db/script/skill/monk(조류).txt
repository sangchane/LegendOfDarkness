0,0,0,0,0,0,0 	SPELL_깃털날리기(Lev2)	{
 	set @myid, get_myid();
	if(get_mapname() == "OX퀴즈장"){message 3, "이벤트공간에서는 사용이불가능합니다."; del @myid; end;}
	if(get_mana(@myid) < 4000){message 3, "마력이 부족합니다."; del @myid; end;}
 	set @x1, get_xs()-1;
 	set @x2, get_xs()+1;
 	set @y1, get_ys()-1;
 	set @y2, get_ys()+1;
	for(set @i, @x1; @i <= @x2; set @i, @i+1){
	for(set @j, @y1; @j <= @y2; set @j, @j+1){
	set @mob, get_mobxy(@i, @j); 
	set @damage, ((get_basevita(@myid)/10));
   	set @type, istype(@mob);
   	if(@mob && @type == 1) {motion 136, 75; effect @mob, 0, 15, 75; game_sound 61, 0; damaged @mob, @damage;}
  		}
 	}
	del @myid;
	del @x1;
	del @x2;
	del @y1;
	del @y2;
	del @mob;
	del @type;
	}
}

0,0,0,0,0,0,0	SPELL_매의위상	{
 set @myid, get_myid();
 if(get_mapname() == "OX퀴즈장"){
 message 3, "이벤트공간에서는 사용이불가능합니다.";	
 }else{
 set @x1, get_xs()-9;
 set @x2, get_xs()+9;
 set @y1, get_ys()-9;
 set @y2, get_ys()+9;       //가로세로범위입니다 8이 화면에 꽉차는거같아보여서..
 set @mana, get_vita(@myid);

 for(set @i, @x1; @i <= @x2; set @i, @i +1){
  for(set @j, @y1; @j <= @y2; set @j, @j+1){
   set @mob, get_mobxy(@i, @j);
   set @damage, ((get_basevita(@myid)+5))*6; //데미지 공식이에요
   set @type, istype(@mob);
   if(@mob && @type == 1 && get_mana(@myid) >= 15500) {                        //타입이 몹이라면
   manal_del "15500";
    effect @mob, 0, 69, 100;
    game_sound 78, 0;
    damaged @mob, @damage;
   }
  }
}
