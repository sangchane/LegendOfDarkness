			del @mob;
			del @type;

			set @type, istype(@mob);
			if(@mob && @type == 1){
				damaged @mob, @damage;
    				effect @mob, 0, 137, 100;
			}else{
			go:
				if(get_map_pk() && istype(get_char_serial(@myid,@i,@j)) == 3){
					set @mob, get_char_serial(@myid,@i,@j);

						char_damaged @mob, @damage,@damage;
						effect @mob, 0, 137, 100;
					}
				}
			}
		}
	}
}
