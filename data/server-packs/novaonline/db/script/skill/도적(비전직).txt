0,0,0,0,0,0,0	SKILL_찌르기	{
	set @myid, get_myid();
	set @target, skill_target();
	set @damage, ((get_dex(@myid))) + 51;
	if(enare(@myid,1)) set @damage, @damage + 8;
	skill_delay 2;
	if(get_mapname() == "OX퀴즈장"){message 3, "이벤트공간에서는 사용이불가능합니다."; end;}
	if(!@target){
		if(istype(get_front_char(@myid)) == 3 && get_map_pk()){
			set @target, get_front_char(@myid);
			char_damaged @target, @damage, @damage;
			motion 135, 20;
			effect @target, 0, 26, 100;
			game_sound 7, 0;
		}
	}else{
		damaged @target, @damage;
		motion 135, 20;
		effect @target, 0, 26, 100;
		game_sound 7, 0;
	}
}
