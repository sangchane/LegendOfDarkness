0,0,0,0,0,0,0	엘리멘탈선물줘용	{
	re:
	set @myid, get_myid();
	del @val;
	set @val, menu("선택해주세요.", "선상교환방법", "선상교환하기");
	if(@val == 1){mes 1, "엘리멘탈사냥터에서 몬스터들이 떨구는 속성의보석 20개를 가져오시면 교환해드려요＊^^＊"; goto re;}
	if(@val == 2){
		if(item_exist(@myid, "속성의보석") < 20){mes 1, "속성의보석 20개가 없습니다."; goto re;}
		item_del "속성의보석", 20;
		item_add "선물상자", 1;
		mes 1, "속성의보석 20개를 선물상자로 교환완료됬어요^^";
		goto re;
	}
	goto re;
}
