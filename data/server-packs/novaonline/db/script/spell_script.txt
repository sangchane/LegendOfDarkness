0,0,0,0,0,0,0	SPELL_쿠로토	{
	set @myid, get_myid();
	if(get_mana(@myid) < 50){message 3, "마력이 부족합니다. [필요마나 : 50]"; del @myid; end;}
	manal_del "50";
	RE:
	set @rand, rand(1,100);
	if(@rand < 1 || @rand > 100){Goto RE;}
	set @hill, 100 + @rand;
	set @type, istype(@myid);
	motion 136, 75;
	effect @target, 4, 0, 75;
	game_sound 8 ,0;
	set_vita @myid, get_vita(@myid)+@hill;
	del @myid;
	del @type;
	del @target;
}
