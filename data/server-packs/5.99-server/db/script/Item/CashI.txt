0,0,0,0,0,0,0	칭호의인장	{
	set @myid, get_myid();
	set @select, 0;
	re:
	set @select, menu("칭호의 인장을 사용하여서 칭호를 장식 하겠습니까?", "장식한다.", "장식하지 않는다.");

	if(@select == 1){
		input @set_user_title$, 1, "[" + get_name() + "]님, 원하는 칭호를 써주십시오.\\n색깔표시(예 : { = l)금지.", "", "", 255;
	  	set_user_title @myid, @set_user_title$;
		item_del "칭호의인장", 1;
		broadcast 5, "{=w[" + get_name() + "]님이 [" + @set_user_title$ + "]칭호를 다셧습니다!!";
		mes 0, "[" + @set_user_title$ + "]칭호를 다셧습니다.";
		end;
	}else if(@select == 2){
		mes 0, "등록을 하지 않고 종료합니다.";
		end;
	}
}

	set @myid, get_myid();
	set @select, 0;
	mes 1, "{=w남자는 여자로, 여자는 남자로 성별을 전환시켜 드립니다.";
	mes 1, "{=w성별전환을 할시 장착중인 아이템을 모두 벗어주세요.";
	if(get_ac(@myid) != 100){mes 0, "장착중인 아이템을 모두 벗어주시길 바랍니다."; end;}
	re:
	set @select, menu("캐릭터의 성별을 전환 하시겠습니까?", "전환한다.");
	if(@select == 0){goto re;}
	if(@select == 1){
		if(get_sex(@myid) == 1){
			set_sex 2;
			set_body @myid, 32;
		}else if(get_sex(@myid) == 2){
			set_sex 1;
			set_body @myid, 16;
		}else{mes 0, "당신의 성별이 불투명합니다.\\n운영자에게 문의해주시길 바랍니다."; end;}
		set_xs(get_xs(@myid));
		item_del "성별전환스크롤", 1;
		mes 0, "당신의 성별이 성공적으로 변경되었습니다.\\n[캐릭터 이미지가 그대로라면 새로고침을 해주세요.]";
		end;
	}
