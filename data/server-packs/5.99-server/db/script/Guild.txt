0,0,0,0,0,0,0	알프레드	{
START:
	set @myid, get_myid();
	set @menu, menu("무엇을 하시겠습니까?", "길드만들기","길드관리");
	if(@menu == 1){
		if(get_guild_state(@myid)){mes 1,"{=g" +  "이미 가입된 길드가 있습니다."; goto START;}
		if(item_exist(@myid, "왕의문장")< 1) {mes 1, "{=g왕의문장이 필요합니다."; goto START;}
MAKE:
		input @name$, 1, "길드명을 입력해주세요", "", "", 10;
		if(!guild_check(@name$)) {
			mes 1,"{=g" + @name$ + " 길드가 만들어졌습니다.";

			set_guild_master @myid, get_name1(@myid);
			set_guild_state @myid, 3;
			set @text$, get_guild_state(@myid) + "길장확인";
			end;
		}
		mes 1, "{=u이미 존재하는 길드입니다."; goto MAKE;
	}else if(@menu == 2){
		if(get_guild_state(@myid) == 0) {mes 1, "{=g" + "가입한 길드가 없습니다."; goto START;}
//일반길드원에 대한 메뉴
		if(get_guild_state(@myid) ==1){set @val, menu("길드원님 반갑습니다. 무엇을 하시겠습니까?","길드원정보","길드탈퇴", "침략");
		if(@val == 1){guild_list(); end;}
		if(@val == 2){
			set @exit, menu("{=u정말 탈퇴하시겠습니까?[탈퇴하실려면 50만골드가 필요합니다.]","예","아니오");
			if(get_money() < 500000){mes 1, "길드탈퇴 실패. 50만 골드가 필요합니다."; end;}
			if(@exit != 1){goto start;}
			money_del "500000";
			set_guild_title @myid, " ";
			set_guild_state @myid, 0; mes 0,"{=g 탈퇴했습니다.";
			end;
		}
	}
//관리창-------------------------------
 MANAGEMENT:
	set @val, menu("반갑습니다. 무엇을 하시겠습니까?","길드원정보","길드원가입","길드원축출","길드해체","길드장이임", "길드탈퇴", "침략");
	if(@val == 1){guild_list();end;}
		
	if(@val == 2){INVITE: input @name$, 1, "초대할 유저 아이디 입력해주세요", "", "", 10; set @target, get_user(@name$); 
	if(!@target) {mes 1,"{=g" +  "존재하지 않는 유저입니다."; goto EXIT;} if(@target == @myid) {mes 1,"{=g" +  "자신은 초대할수 없습니다."; goto INVITE;}
	if(get_mapname1(@myid) != get_mapname1(@target)) {mes 1,"{=g" +  "같은 맵에 존재해야합니다."; goto INVITE;}
	if(get_guild_state(@target)) {mes 1,"{=g" + get_name1(@target) + "님은 이미 가입된 길드가 있습니다."; goto INVITE;}
	set @sel, menu("{=g" + get_name1(@target) + "님을 길드에 초대하시겠습니까?", "네 초대합니다.", "아니요");
	if(@sel == 1) {message1 @target ,3,"길드초대장이 전달받았습니다."; mes 1,"{=g" +  get_name1(@target) + " 님께 길드초대장을 보냈습니다. 답장이 올때까지 대기하세요.\\n" + get_name1(@target) + "님께서 초대에 응했을때 길드가입이 이루어집니다."; item_add1 @target, "길드초대장", 1; set_guild_master @target, get_guild_master(@myid);} goto MANAGEMENT;}
	
	if(@val == 3) {EXIT: input @name$, 1, "축출할 유저 아이디 입력해주세요", "", "", 10; set @target, get_user(@name$);
	if(!@target) {mes 1,"{=g" +  "존재하지 않는 유저입니다."; goto EXIT;}
	if(@target == @myid) {mes 1,"{=g" +  "자신은 강퇴할수 없습니다."; goto EXIT;}
	if(get_guild_master(@myid) != get_name1(@myid) && get_guild_state(@target) >= 2) {mes 1, "{=g 간부진에 대한 처리는 길드장만이 가능합니다."; goto MANAGEMENT;}
	if(!get_guild_state(@target)) {mes 1,"{=g" + get_name1(@target) + "님은 가입된 길드가 없습니다."; goto EXIT;}
	if(get_guild_master(@target) != get_guild_master(@myid)) {mes 1, "{=g 같은 길드가 아닙니다."; goto EXIT;}
	set @sel, menu("{=g" + get_name1(@target) + "님을 길드에서 축출시키겠습니까?", "네", "아니요");
	if(@sel == 1) {set_guild_master @target, " "; set_guild_title @target, " "; set_guild_state @target, 0; message1 @target, 3, "{=g당신은 " + get_name() + " 님에 의해서 길드에서 축출당했습니다."; message1 @myid, 3, "{=g" + get_name1(@target) + "님을 축출시켰습니다.";}goto MANAGEMENT;}
	
	if(@val == 4){
	if(get_guild_state(@myid) == 3) {
	if(guild_count(0) == 1) {set @sel, menu("길드마스터가 탈퇴하면 길드는 해체됩니다. 정말 탈퇴하시겠습니까?","예","아니오");
	if(@sel == 1) {message1 @myid,3 ,"{=c길드가 해체되었습니다"; set_guild_master @myid, " "; set_guild_state @myid, 0; set_guild_title @myid, " "; end;}
	else{goto MANAGEMENT;}
	}
	mes 1,"{=g길드마스터의 탈퇴는 길드해체와 같습니다. 마스터 혼자 길드에 남아있을때 가능합니다. 남아있는 길드원이 있는지 확인후 다시 시도해주십시오."; goto MANAGEMENT;
	}
	}else if(@val == 5){
LEAVE:			if(get_guild_state(@myid) != 3) {mes 1, "{=g 길드장만이 가능합니다."; goto MANAGEMENT;}
		input @name$, 1, "이임할 유저 아이디 입력해주세요", "", "", 10;
		
		set @target, get_user(@name$);
		if(!@target) {mes 1,"{=g" +  "존재하지 않는 유저입니다."; goto LEAVE;}
		if(@target == @myid) {mes 1,"{=g" +  "자신은 할 수 없습니다."; goto LEAVE;}
		if(!get_guild_state(@target)) {mes 1,"{=g" + get_name1(@target) + "님은 가입된 길드가 없습니다."; goto LEAVE;}
		if(get_guild_title(@target) != get_guild_title(@myid)) {mes 1, "{=g 같은 길드가 아닙니다."; goto LEAVE;}
		
		mes 1, "길드장 이임 후에 자신은 부길드장 권한을 갖습니다.";
		set @name$, @name$ + "님께 길드장이임 하시겠습니까? 신중한 결정 부탁드립니다.";
		set @sel, menu(@name$,"네","아니오");
		if(@sel == 1) {
			set_guild_state @myid, 2;
			set_guild_state @target, 3;
			set_guild_master @myid, get_name1(@target);
			set_guild_master @target, get_name1(@target);
			message1 @target, 3, "{=u" + get_guild_title(@myid) + " 길드의 길드장이 되셨습니다.";
		}else{goto MANAGEMENT;}
		end;
	}else if(@val == 6){
SECOND:			if(get_guild_state(@myid) != 3) {mes 1, "{=g 길드장만이 가능합니다."; goto MANAGEMENT;}
			input @name$, 1, "위임할 유저 아이디 입력해주세요", "", "", 10;
		
		set @target, get_user(@name$);
		if(!@target) {mes 1,"{=g" +  "존재하지 않는 유저입니다."; goto SECOND;}
		if(@target == @myid) {mes 1,"{=g" +  "자신은 할 수 없습니다."; goto SECOND;}
		if(!get_guild_state(@target)) {mes 1,"{=g" + get_name1(@target) + "님은 가입된 길드가 없습니다."; goto SECOND;}
		if(get_guild_master(@target) != get_guild_master(@myid)) {mes 1, "{=g 같은 길드가 아닙니다."; goto SECOND;}
		
		set @name$, "{=u"+ @name$ + "님께 부길드장 권한을 부여합니다. 동의하십니까?";
		set @sel, menu(@name$,"네","아니오");
		if(@sel == 1) {
			set_guild_state @target, 2;
			message1 @target,3, "{=u" + get_guild_title(@myid) + " 길드의 부길드장이 되셨습니다. 길드관리메뉴 이용가능.";
		}else{goto MANAGEMENT;}
		end;
	}else if(@val == 7){
		if(get_guild_state(@myid) != 2) {mes 1, "{=g 길드장은 탈퇴가 불가능합니다."; goto MANAGEMENT;}
		set @exit, menu("{=u정말 탈퇴하시겠습니까?[탈퇴하실려면 1백만골드가 필요합니다.]","예","아니오");
		if(@exit == 1){
			if(get_money() < 1000000){mes 1, "길드탈퇴 실패. 1백만 골드가 필요합니다."; end;} 
			money_del "1000000";
			set_guild_title @myid, " ";
			set_guild_state @myid, 0;
			mes 0,"{=g 탈퇴했습니다.";
			end;
		}else{goto MANAGEMENT;}
	}
		
	}
}

		set_guild_title @myid, get_guild_title(get_user(get_guild_master(@myid)));

 		if(!get_user(get_guild_master(@myid))) {item_del "길드초대장", 1; end;}

 		set_guild_master @myid, " ";
