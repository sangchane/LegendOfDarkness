0,0,0,0,0,0,0	SKILL_M포효	{
	set @myid, get_myid();
	if(get_mana(@myid) < get_basemana(@myid)/100*3){message 3, "마력이 부족합니다. [필요마력 : 3%]"; del @myid; end;}
   	manal_del get_basemana(@myid)/100*3;
	set @x1, get_xs(@myid)-7;
	set @x2, get_xs(@myid)+7;
	set @y1, get_ys(@myid)-7;
	set @y2, get_ys(@myid)+7;
	for(set @i, @x1; @i <= @x2; set @i, @i +1){
		for(set @j, @y1; @j <= @y2; set @j, @j+1){
			del @mob;
			del @type;
			set @mob, get_mobxy(@i, @j);
			if(!@mob){goto go;}
			set @type, istype(@mob);
			if(@mob && @type == 1){
				magic 6,@mob, 0, 13, 0,@myid;
    				effect @mob, 0, 41, 75;
			}
			go:
		}
	}
	message 3, "M포효를 시전하셧습니다.";
	effect @myid, 73, 0, 100;
   	game_sound 48, 0;
	skill_delay 22;
	del @val;
}
