	message 3, "메테오를 외웠습니다.";
   	game_sound 63, 0;
	set @damage, get_mana(@myid) / 4 * 10;
	set @x1, get_xs(@myid)-7;
	set @x2, get_xs(@myid)+7;
	set @y1, get_ys(@myid)-7;
	set @y2, get_ys(@myid)+7;
	for(set @i, @x1; @i <= @x2; set @i, @i +1){
		for(set @j, @y1; @j <= @y2; set @j, @j+1){
			del @mob;
			del @type;
			set @mob, get_mobxy(@i, @j);
			if(!@mob){goto go;}
			set @type, istype(@mob);
			if(@mob && @type == 1){
				effect @mob, 0, 72, 100;
				effect @mob, 0, 72, 100;
				damaged @mob, @damage;
			}else{
			go:
				if(get_map_pk() && istype(get_char_serial(@myid,@i,@j)) == 3){
					set @damage, get_mana(@myid);
					set @mob, get_char_serial(@myid,@i,@j);
					if(@myid != @mob){
						effect @mob, 0, 72, 100;
						char_damaged2 @mob, @damage,@damage;
					}
				}
			}
		}
	}
	set_manal 0;
}
