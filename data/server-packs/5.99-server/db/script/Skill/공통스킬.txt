0,0,0,0,0,0,0	SPELL_로카메아	{
	set @myid, get_myid();
	set @name$, spell_input(@myid);
	set @target, get_user(@name$);
	if(!@target){message 3, "대상이 존재하지 않습니다."; end;}
	if(@target == @myid){message 3, "자기자신을 소환대상으로 정할수는 없습니다."; end;}
	if(get_mapname1(@target) != get_mapname1(@myid)){message 3, "같은 맵에 있는 유저만 소환이 가능합니다."; end;}
	message 3, "[" + @name$ + "]님을 소환합니다.";
	warp3 @target, get_mapname1(@myid), get_xs(@myid), get_ys(@myid);
	message1 @target, 3, get_name(@myid) + "님이 로카메아로 당신을 소환했습니다.";
}
