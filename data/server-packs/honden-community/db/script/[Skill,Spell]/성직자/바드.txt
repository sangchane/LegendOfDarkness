0	SKILL_연주공격(Lev1)	{
	set@myid,get_myid();
	set@target,skill_target();
	if(get_mana(@myid)<get_basemana(@myid)/100*2){message 3,"마력이 부족합니다.";end;}
	set_manal get_mana(@myid) - get_basemana(@myid)/100*1;
	set@x1,get_xs();
	set@y1,get_ys();
	set@dam,(get_basemana2(@myid)*1)/100*40;
	set @i, 0;
	set @y, 0;
	set @damage, 0;
	set @sex, 0;
	if(get_basemana2(@myid) <= 100000){
		set@dam,@dam/100*112;
	}else{
		for(set @i, @i;  @i <= 100; set @i, @i+1){
			set @sex, @sex+1;		
			if(100000 +(@sex * 10000) >  get_basemana2(@myid)){
				set @dam, @dam / 100 * (112 + @sex);
				break;
			}
		}
	}
	set@side,get_side(@myid);
	set@type,istype(@target);
	set@side,get_side(@myid);
	if(get_map_pk()){
		switch(@side){
		case 0:
		set@mob1,get_char_serial(@myid,@x1,(@y1)-1);
		set@mob2,get_char_serial(@myid,@x1,(@y1)-2);
		set@mob3,get_char_serial(@myid,@x1,(@y1)-3);
		if(istype(@mob1)==3){char_damaged@mob1,@dam,0,0;}
		if(istype(@mob2)==3){char_damaged@mob2,@dam,0,0;}
		if(istype(@mob3)==3){char_damaged@mob3,@dam,0,0;}
		break;
		case 1:
		set@mob1,get_char_serial(@myid,(@x1)+1,@y1);
		set@mob2,get_char_serial(@myid,(@x1)+2,@y1);
		set@mob3,get_char_serial(@myid,(@x1)+3,@y1);
		if(istype(@mob1)==3){char_damaged@mob1,@dam,0,0;}
		if(istype(@mob2)==3){char_damaged@mob2,@dam,0,0;}
		if(istype(@mob3)==3){char_damaged@mob3,@dam,0,0;}
		break;
		case 2:
		set@mob1,get_char_serial(@myid,@x1,(@y1)+1);
		set@mob2,get_char_serial(@myid,@x1,(@y1)+2);
		set@mob3,get_char_serial(@myid,@x1,(@y1)+3);
		if(istype(@mob1)==3){char_damaged@mob1,@dam,0,0;}
		if(istype(@mob2)==3){char_damaged@mob2,@dam,0,0;}
		if(istype(@mob3)==3){char_damaged@mob3,@dam,0,0;}
		break;
		case 3:
		set@mob1,get_char_serial(@myid,(@x1)-1,@y1);
		set@mob2,get_char_serial(@myid,(@x1)-2,@y1);
		set@mob3,get_char_serial(@myid,(@x1)-3,@y1);
		if(istype(@mob1)==3){char_damaged@mob1,@dam,0,0;}
		if(istype(@mob2)==3){char_damaged@mob2,@dam,0,0;}
		if(istype(@mob3)==3){char_damaged@mob3,@dam,0,0;}
		break;
		}

	direct_damage@myid,0,3,@dam,0,0;
	motion 137,20;
	game_sound 1,0;
	skill_delay 1;
	del@dam;
	del@mob1;
