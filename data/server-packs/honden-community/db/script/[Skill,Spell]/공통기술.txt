0	SKILL_채집	{
	set @myid, get_myid();
	set@time,time() - #gathering;
	if(get_wearitem_weapon(@myid) != "채집망"||get_wearitem_weapon(@myid) == 0){message 3, "[채집]에 필요한 장비가 없습니다."; end;}
	if(@time<4){end;}
	skill_delay 4;
	set @side, get_side(@myid);
	switch(@side){
		case 0:
			if(get_xy_block(@myid, get_xs(@myid), get_ys(@myid)-1)){goto AA;}
		break;
		case 1:
			if(get_xy_block(@myid, get_xs(@myid)+1, get_ys(@myid))){goto AA;}
		break;
		case 2:
			if(get_xy_block(@myid, get_xs(@myid), get_ys(@myid)+1)){goto AA;}
		break;
		case 3:
			if(get_xy_block(@myid, get_xs(@myid)-1, get_ys(@myid))){goto AA;}
		break;
	}
	message 3,"곤충 채집이 불가능한 구역입니다.";
	end;
	AA:
	if(item_exist(@myid, "채집용꿀")||item_exist(@myid, "채집용설탕물")){goto go1;}
	message 3, "미끼가 없습니다";
	end;
	go1:
	set #gathering,time();
	if(!item_exist(@myid, "채집용설탕물")){
		item_del "채집용꿀", 1;
		set #case6, 1;
	}else{
		item_del "채집용설탕물", 1;
		set #case6, 0;
	}
	if(#case6==1){
		set #case6, 0;
		if(rand2(1,100) > 50){
			goto case1;
		}else{goto case2;}
	}
	if(rand2(1,100) > 5){
		case1:
		message 3, "미끼만 날렸습니다.";
	}else{
		re2:
		set @rand, rand2(1,3);
		if(@rand < 1 || @rand > 3){goto re2;}
		switch(@rand){
			case 1: set @item$, "세토아의가호(Lev1)"; break;
			case 2: set @item$, "세토아의가호(Lev2)"; break;
			case 3: set @item$, "이아의가호(Lev1)"; break;
		}
		item_add @item$, 1;
		message 3, "" + @item$ + "를 잡았습니다.";
		goto R;
		case2:
		re3:
		set @rand, rand2(1,5);
		if(@rand < 1 || @rand > 5){goto re3;}
		switch(@rand){
			case 1: set @item$, "세토아의가호(Lev1)"; break;
			case 2: set @item$, "세토아의가호(Lev2)"; break;
			case 3: set @item$, "이아의가호(Lev1)"; break;
			case 4: set @item$, "이아의가호(Lev2)"; break;
			case 5: set @item$, "세오의가호(Lev1)"; break;
		}
		item_add @item$, 1;
		message 3, "" + @item$ + "를 잡았습니다.";
		goto R;
	}
	R:
	motion 1, 25;
	game_sound 9, 0;
	del @rand;
	del @item$;
	end;
}
